from fastapi import FastAPI, UploadFile, File
import os
import shutil
from services.video import render_portrait_clip

app = FastAPI(title="AI Video Clipper Engine")

os.makedirs("uploads", exist_ok=True)
os.makedirs("outputs", exist_ok=True)

@app.post("/api/render-clip")
async def render_clip_api(file: UploadFile = File(...), start: float = 0.0, end: float = 10.0, center_x: int = 500):
    input_path = f"uploads/{file.filename}"
    with open(input_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
        
    output_path = f"outputs/portrait_{file.filename}"
    
    try:
        render_portrait_clip(input_path, start, end, center_x, output_path)
        return {
            "status": "success",
            "message": "Klip berhasil dipotong ke format 9:16!",
            "download_url": output_path
        }
    except Exception as e:
        return {"status": "error", "message": str(e)}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=10000)
